# Ask ChatGPT for any issues
# Steps to run the coresets pipeline on EC2 machine
# Convert this into a CI GH Action #TODO
# Change permissions on your local machine
chmod 400 <pem_file>

# Login to EC2
ssh -i <pem_file> ubuntu@<public-ip>

# Check installations and version
git --version
python3 --version # use python 3.12

# Run ubuntu specific commands
sudo apt update
sudo apt install -y python3.12 python3.12-venv git python3-pip
sudo apt install unzip

# Check S3 listing access, if access issue then reach out to AWS team and share the error with them
aws s3 ls s3://<container-name>/processed_dataset/curriculum_pyspark_output/source=flan/

# Install AWS CLI if previous command did not work due to awscli not present in the OS
curl "https://awscli.amazonaws.com/awscli-exe-linux-x86_64.zip" -o awscliv2.zip
unzip awscliv2.zip
sudo ./aws/install
rm -rf aws awscliv2.zip

# Required to run only when new EC2 machine
# Set Git username and email (Git does NOT store passwords)
git config --global user.name "[YOUR_USERNAME]"
git config --global user.email "[EMAIL_ADDRESS]"

# Clone repository
# First check if repo already exists or not
ls -lrt LLM/
# Run if previous command fails
git clone https://github.com/The-School-of-AI/LLM.git
cd LLM

# Checkout a specific remote branch
git fetch origin
git checkout -b p3/feat/stage-wise-coreset-selection_v2 origin/p3/feat/stage-wise-coreset-selection_v2

# Push to changes from local to the branch and repull in the EC2 instance
git pull origin p3/feat/stage-wise-coreset-selection_v2

# Create virtual environment and install dependencies
python3 --version
python3.12 -m venv venv
source venv/bin/activate
pip install -U pip setuptools wheel
pip install pyyaml numpy xxhash scipy pandas boto3 awscli pyarrow

# Install dependencies
pip install -r experiments/3_coreset_engineering/coreset_engine_v5/requirements.txt

# One can also use tmux to run this command in the backend.
# Tmux is used to run the process in background on Ubuntu
# One can directly call the shard.sh from bash as well
tmux new -s coreset
source venv/bin/activate
python --version
nohup bash experiments/3_coreset_engineering/coreset_engine_v5/shard.sh \
  --num-shards 8 \
  --stages "1B" \
  --input-path "s3://<container-name>/processed_dataset/curriculum_pyspark_output/" \
  --input-format jsonl \
  --total-tokens 4523096944 \
  --resume \
  > shard_run.log 2>&1 &

ps aux | grep shard.sh
tail -f shard_run.log

# To come out of Tmux session (detach only not kill)
Control + B, then D

# Use this to see the process logs
tail -f shard_run.log

# To kill the session
tmux kill-session -t coreset

aws s3 sync /home/ubuntu/LLM/experiments/3_coreset_engineering/coreset_engine_v5/output/ s3://<container-name>/coreset_outputs/run_2/
