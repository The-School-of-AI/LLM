#!/bin/bash
# EMR Deployment for FAST T2 Curriculum Calculator
# Expected cost: $50-100 total (10-20x cheaper than regex approach)
# Expected time: 12-24 hours for all datasets

set -e

echo "=== T2 Fast Curriculum - EMR Deployment ==="

# Configuration - UPDATE THESE
CLUSTER_NAME="T2-Fast-Curriculum"
KEY_NAME="your-ec2-key"               # UPDATE
SUBNET_ID="subnet-xxxxxxxx"           # UPDATE
S3_BUCKET="your-bucket-name"          # UPDATE
S3_SCRIPT="s3://${S3_BUCKET}/scripts/t2_fast_curriculum.py"
S3_LOGS="s3://${S3_BUCKET}/emr-logs/"

# Upload script
echo "Uploading fast curriculum script..."
aws s3 cp t2_fast_curriculum.py $S3_SCRIPT

# Create Spark config (much lighter than before)
cat > spark-config.json << 'EOF'
[
  {
    "Classification": "spark-defaults",
    "Properties": {
      "spark.executor.cores": "4",
      "spark.executor.memory": "12g",
      "spark.executor.memoryOverhead": "2g",
      "spark.driver.memory": "12g",
      "spark.driver.cores": "4",
      "spark.sql.adaptive.enabled": "true",
      "spark.sql.adaptive.coalescePartitions.enabled": "true",
      "spark.sql.files.maxPartitionBytes": "268435456",
      "spark.default.parallelism": "200",
      "spark.dynamicAllocation.enabled": "true",
      "spark.dynamicAllocation.minExecutors": "2",
      "spark.dynamicAllocation.maxExecutors": "30"
    }
  }
]
EOF

echo "Creating EMR cluster..."
echo "  Instance type: r5.xlarge (cheaper, sufficient for fast processing)"
echo "  Dynamic allocation: 2-30 executors"
echo ""

CLUSTER_ID=$(aws emr create-cluster \
    --name "$CLUSTER_NAME" \
    --release-label emr-6.15.0 \
    --applications Name=Spark \
    --log-uri $S3_LOGS \
    --ec2-attributes KeyName=$KEY_NAME,SubnetId=$SUBNET_ID \
    --instance-groups '[
        {
            "Name": "Master",
            "InstanceGroupType": "MASTER",
            "InstanceType": "r5.xlarge",
            "InstanceCount": 1,
            "Market": "ON_DEMAND"
        },
        {
            "Name": "Core",
            "InstanceGroupType": "CORE",
            "InstanceType": "r5.xlarge",
            "InstanceCount": 2,
            "Market": "ON_DEMAND"
        },
        {
            "Name": "Task-Spot",
            "InstanceGroupType": "TASK",
            "InstanceType": "r5.2xlarge",
            "InstanceCount": 10,
            "Market": "SPOT",
            "BidPrice": "0.25"
        }
    ]' \
    --configurations file://spark-config.json \
    --service-role EMR_DefaultRole \
    --ec2-attributes InstanceProfile=EMR_EC2_DefaultRole \
    --auto-terminate \
    --query 'ClusterId' \
    --output text)

echo "Cluster ID: $CLUSTER_ID"
echo ""
echo "Waiting for cluster to be ready..."

aws emr wait cluster-running --cluster-id $CLUSTER_ID

echo "✅ Cluster is ready!"
echo ""
echo "Cluster details:"
echo "  ID: $CLUSTER_ID"
echo "  Monitor: https://console.aws.amazon.com/emr/home?region=us-east-1#/clusterDetails/$CLUSTER_ID"
echo ""
echo "Next steps:"
echo "  1. Test with failed datasets: ./test_on_emr.sh $CLUSTER_ID"
echo "  2. Run production: ./run_production_fast.sh $CLUSTER_ID"
